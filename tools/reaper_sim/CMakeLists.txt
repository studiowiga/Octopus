# Test du pont REAPER contre une API de REAPER simulée (Linux, hors plugin)
cmake_minimum_required(VERSION 3.22)
project(OctopusReaperSim VERSION 1.0.0)
set(CMAKE_CXX_STANDARD 17)
add_subdirectory(${JUCE_DIR} JUCE)
set(W ${CMAKE_CURRENT_SOURCE_DIR}/../..)
juce_add_console_app(OctopusReaperSim PRODUCT_NAME "OctopusReaperSim")
target_sources(OctopusReaperSim PRIVATE main.cpp ${W}/ReaperBridge.cpp ${W}/OctopusEngine.cpp)
target_include_directories(OctopusReaperSim PRIVATE ${W} ${JUCE_DIR}/modules/juce_audio_processors_headless/format_types/VST3_SDK)
target_compile_definitions(OctopusReaperSim PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_STANDALONE_APPLICATION=1)
target_link_libraries(OctopusReaperSim PRIVATE juce::juce_graphics juce::juce_data_structures juce::juce_recommended_config_flags)
